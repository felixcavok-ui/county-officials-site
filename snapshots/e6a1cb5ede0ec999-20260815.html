/*

 读取方法 getCookie(存储名称) 例 getCookie("mode"); 
 设置方法 setCookie(名称:自定义参数:过期时间) 例 setCookie("mode", "true", "d1"); 
*/
function setCookie(name, value, time) {
 var strsec = getsec(time);
 var exp = new Date();
 exp.setTime(exp.getTime() + strsec * 1);
 document.cookie = name + "=" + escape(value) + ";path=/;expires=" + exp.toGMTString();
}
function getsec(str) {
 var str1 = str.substring(1, str.length) * 1;
 var str2 = str.substring(0, 1);
 if (str2 == "s") {
 return str1 * 1000;
 } else if (str2 == "h") {
 return str1 * 60 * 60 * 1000;
 } else if (str2 == "d") {
 return str1 * 24 * 60 * 60 * 1000;
 }
}
function getCookie(name) {
 var arr, reg = new RegExp("(^| )" + name + "=([^;]*)(;|$)");
 if (arr = document.cookie.match(reg)) return unescape(arr[2]);
 else return null;
}
function delCookie(name) {
 var exp = new Date();
 exp.setTime(exp.getTime() - 1);
 var cval = getCookie(name);
 if (cval != null)
 document.cookie = name + "=" + cval + ";path=/;expires=" + exp.toGMTString();
}

 各乡镇人民政府，诏安工业园区管委会，县直各单位： 经研究，现将县人民政府领导成员工作分工如下： 一、张延能代县长（县政府党组书记）：主持县政府全面工作。负责审计工作。 分管部门： 县审计局。 二、林文绍常委、副县长（县政府党组成员）：协助郭景宏副县长抓工信、商务、科技、招商引资、行政服务、央企民企对接、诏安工业园区等工作。 三、蔡燕海常委、副县长（县政府党组副书记）：负责县政府常务工作；县长外出时，主持县政府日常工作。负责发改（粮食和物资储备、数据）、统计、交通运输（铁路）、应急管理（防汛抗旱）、地震、退役军人事务、国防动员、人防、机关效能、对口支援、区域经济协作、山海协作等工作。协助县长联系审计工作。 分管部门： 县政府办（效能办、国动办、人防办、县政府外事办公室、县政府港澳事务办公室）、发改局（粮储局、数据管理局，重点项目办）、统计局、交通运输局（交通运输综合执法大队、运输综合服务中心，铁路办）、应急局（应急管理综合执法大队，安委办）、退役军人局（双拥办），地震办。 联系单位： 县人大常委会，县委网信办、保密委员会办公室（县国家保密局）、机要局（县密码管理局），县审计局（经济责任办），县委县政府接待服务中心，农村社会经济调查队、城市社会经济调查队，消防救援大队（消防救援局，消安委办），漳州公路诏安分中心，漳州车务段诏安站，武警中队、海警工作站，驻诏部队。 四、朱添泉常委（县政府党组成员）：负责自然资源、住建、住房公积金管理等工作；协管招商引资工作。 分管部门： 县自然资源局（不动产登记中心、城乡规划中心）、住建局（住房保障和房地产服务中心）。 联系单位： 漳州市住房公积金中心诏安管理部。 五、黄海强副县长（县政府党组成员）：负责公安、司法、信访等工作；兼任县信访联席会议召集人。 分管部门： 县公安局（打私办）、司法局、信访局，道安办。 联系单位： 县法院、检察院，县国家安全局。 六、林海亮副县长（县政府党组成员）：协助蔡燕海常委、副县长抓发改（粮食和物资储备、数据）、统计、交通运输（铁路）、应急管理（防汛抗旱）、地震、退役军人事务、国防动员、人防、机关效能、对口支援、区域经济协作、山海协作等工作。 七、吕自信副县长（县政府党组成员）：负责农业农村、乡村振兴、水利、林业、海洋渔业、生态环保、供销等工作。 分管部门： 县农业农村局（乡村振兴局，农业综合执法大队、水果开发中心、良种场、农业机械服务中心，富硒办）、水利局（库区移民开发中心）、林业局、海洋渔业局（海洋与渔业服务中心），供销联社。 联系单位： 诏安生态环境局（环境监测站）、气象局。 八、郭景宏副县长（县政府党组成员）：负责工信、商务、科技、招商引资、行政服务、央企民企对接、诏安工业园区等工作。 分管部门： 县工信局（商务局、科技局，招商服务中心），行政服务中心管委会，二轻联社，诏安工业园区管委会。 联系单位： 县工商联、科协，东山海关，烟草专卖局、盐务局，供电公司、邮政公司、电信公司、移动公司、联通公司、铁塔公司。 九、许逸炫副县长（县政府党组成员）：负责财税、国资、县属国有企业、金融、民政、老区、老龄、残疾人事业、民族与宗教事务、人力资源和社会保障等工作。 分管部门： 县财政局（国资办），金融办，民政局（老区办）、人社局（退休干部工作站）。 联系单位： 县委编办，县民族宗教局，残联，老龄委，诏安县税务局、诏安金融监管支局，在诏金融机构。 十、王惠敏副县长：负责教育、卫生健康、医保、市场监督管理（药品监督管理工作）、妇女儿童等工作。 分管部门： 县教育局（县政府教育督导室）、卫健局（疾病预防控制局），总医院、妇幼保健院、疾控中心（卫生健康监督所）。 联系单位： 县政协，老促会、老体协、老年大学、关工委，妇联，慈善总会，计生协会、红十字会，诏安医保局。 十一、蓝志清副县长（县政府党组成员）：负责文化体育旅游（文物）、城管执法、市场监管、广播电视、侨台、政府系统精神文明建设等工作。 分管部门： 县文旅局（新闻出版广电局，广播电视中心、旅游服务中心、文化市场综合执法大队）、城管执法局（城市管理和综合执法大队、市容服务中心）、市场监管局（知识产权局，质量计量检验检测站，除药品监督管理工作外）。 联系单位： 县委台办（县政府台湾事务办公室）、老干局、党校、文明办、史志室，县侨办，侨联、台联、团县委，档案馆，文联，社科联，五Ο五台，融媒体中心，广电网络公司。 十二、康德坤四级调研员：协助王惠敏副县长开展工作。 联系单位： 诏安县企业与企业家联合会。 十三、林燕坤（县政府党组成员、县政府办主任）：协助处理县政府日常工作，负责县长交办的其他事项。主持县政府办公室全面工作。   备注： 为保障县政府工作高效有序开展，领导成员之间互相补台，特设立县政府领导成员AB岗。具体如下： 林文绍同志与郭景宏同志互为AB岗； 蔡燕海同志与林海亮同志互为AB岗； 朱添泉同志与黄海强同志互为AB岗； 吕自信同志与王惠敏同志互为AB岗； 许逸炫同志与蓝志清同志互为AB岗。 诏安县人民政府 2026年7月24日
