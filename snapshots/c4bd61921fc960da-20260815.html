function setCookie(name, value, time) {
 var strsec = getsec(time);
 var exp = new Date();
 exp.setTime(exp.getTime() + strsec * 1);
 document.cookie = name + "=" + escape(value) + ";path=/;expires=" + exp.toGMTString();
}
function getsec(str) {
 var str1 = str.substring(1, str.length) * 1;
 var str2 = str.substring(0, 1);
 if (str2 == "s") {
 return str1 * 1000;
 } else if (str2 == "h") {
 return str1 * 60 * 60 * 1000;
 } else if (str2 == "d") {
 return str1 * 24 * 60 * 60 * 1000;
 }
}
function getCookie(name) {
 var arr, reg = new RegExp("(^| )" + name + "=([^;]*)(;|$)");
 if (arr = document.cookie.match(reg)) return unescape(arr[2]);
 else return null;
}
function delCookie(name) {
 var exp = new Date();
 exp.setTime(exp.getTime() - 1);
 var cval = getCookie(name);
 if (cval != null)
 document.cookie = name + "=" + cval + ";path=/;expires=" + exp.toGMTString();
}

　　近日，东山县县长黄键鹏带领相关部门人员深入陈城镇绿秋蔬菜种植家庭农场、西埔供销社农资配送中心检查指导春耕备耕工作。县领导朱叶顺参加检查指导。
　　
每到一处，黄键鹏都详细询问今年春耕农资、肥料农药和种子储备、价格行情等情况，并认真听取了相关负责人关于春耕备耕生产和农资供销的情况汇报，现场协调解决相关问题。黄键鹏指出，要把疫情防控作为春耕生产的前提和基础，教育引导农民和各类经营主体有序下田、分时下地，避免人员聚集，保障农民安全春耕。
黄键鹏强调，当前正值疫情防控的特殊时期，也是春耕备耕的关键时期，各级各部门要按照中央、省、市、县有关部署要求，加强组织领导，强化要素保障，严格落实防护措施，科学安全实现春耕生产；相关部门要进一步落实各项强农惠农政策，确保今年春耕期间农资供应充足、质量可靠、价格稳定；要在做好疫情防控的基础上，不误农时，协调有关部门，为农民群众在农田灌溉、农资物品供应、农业技术等方面提前做好服务工作，做到疫情防控和春耕生产统筹推进。
