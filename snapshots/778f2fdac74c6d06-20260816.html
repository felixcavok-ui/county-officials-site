//在网页中跟随滚动条上下移动的类

function floatlayer(layerid)

{

 this.lastScrollY=0;

 this.layerid=layerid;

 this.backtop=function()

 {

 try

 {

 var scrollPos; 

 if (typeof window.pageYOffset != 'undefined') { 

 scrollPos = window.pageYOffset; 

 } 

 else if (typeof document.compatMode != 'undefined' && document.compatMode != 'BackCompat') { 

 scrollPos = document.documentElement.scrollTop; 

 } 

 else if (typeof document.body != 'undefined') { 

 scrollPos = document.body.scrollTop; 

 } 

 diffY=scrollPos;

 percent=.1*(diffY-this.lastScrollY);

 if(percent>0)

 {

 percent=Math.ceil(percent);

 }

 else

 {

 percent=Math.floor(percent);

 }

 document.getElementById(this.layerid).style.top=parseInt(document.getElementById(this.layerid).style.top)+percent;

 this.lastScrollY=this.lastScrollY+percent;

 }

 catch(e)

 {

 window.status="层"+this.layerid+"不存在";

 }

 }

}

var floatlayer1=new floatlayer("floater");//layer1是你页面上加的层的名字。要对层设计大小。

window.setInterval("floatlayer1.backtop()",10);
