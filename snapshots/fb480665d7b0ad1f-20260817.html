function setCookie(name, value, time) {
 var strsec = getsec(time);
 var exp = new Date();
 exp.setTime(exp.getTime() + strsec * 1);
 document.cookie = name + "=" + escape(value) + ";path=/;expires=" + exp.toGMTString();
}
function getsec(str) {
 var str1 = str.substring(1, str.length) * 1;
 var str2 = str.substring(0, 1);
 if (str2 == "s") {
 return str1 * 1000;
 } else if (str2 == "h") {
 return str1 * 60 * 60 * 1000;
 } else if (str2 == "d") {
 return str1 * 24 * 60 * 60 * 1000;
 }
}
function getCookie(name) {
 var arr, reg = new RegExp("(^| )" + name + "=([^;]*)(;|$)");

 else return null;
}

                            2005年3月9日在浦城县第十四届 人民代表大会第二次会议上 
 各位代表：
 现在，我代表浦城县人民政府向大会作工作报告，请予审议，并请县政协委员和其他列席人员提出意见。
 一、攻坚克难实现政府工作良好开局的一年
 2004年是新一届县人民政府履行使命和职责的第一年，也是发展进程中不平凡的一年。一年来，县政府在市委、市政府和县委的正确领导下，坚持以“三个代表”重要思想为指导，认真贯彻党的十六大和十六届三中、四中全会精神及省、市的一系列决策部署，全面树立和落实科学发展观，紧紧抓住发展这个第一要务，团结一致，积极应对禽流感威胁，努力克服干旱缺电及经济环境偏紧等困难，在全县人民的共同努力下，顺利完成县十四届人大一次会议确定的各项目标任务。全年实现生产总值23.49亿元，增长10.7%。全社会固定资产投资10.98亿元，增长29.52%。财政总收入1.35亿元，同口径增长10.84%，其中地方级收入8062万元，同口径增长18.5%。社会消费品零售总额8.15亿元，增长13.5%。实际利用外资2320万美元，增长11.9%。外贸出口3153万美元，增长10.94%。城镇居民人均可支配收入7912.5元，增长12.47%；农民人均纯收入3748元，增长11.6%。城镇登记失业率控制在5.6%。人口自然增长率为8.25‰。一年来主要工作成效是：
 一是工业经济质量稳步提升。坚持自力更生和争取上级支持，千方百计化解工业用电、用地和信贷紧缺等困难，强化运行调度，精心组织生产，确保工业经济快速增长。实现工业总产值17.39亿元，增长16.6%，对经济增长的贡献率达23.94 %。规模工业加速扩张，规模以上工业企业达45家，比年初净增12家，培育年产值超亿元2家，超千万元24家；完成规模工业产值11.24亿元，增长24.1%，占全社会工业总产值的64.6%；经济效益综合指数达 166，继续位居全市首位。工业园区一期开发基本完成，新引进9家企业，入园企业达23家，总投资4亿元，完成投资1.8亿元；其中10家投产企业实现产值超亿元，利税1500余万元，安置就业2000多人。民营工业更趋活跃，36家民营规模工业企业完成产值9.16亿元，占规模工业总产值的81.5%，经济自主增长动力进一步增强。
 二是“三农”工作扎实有效。坚持把“三农”工作摆在全局工作首位，全面落实支农扶农强农各项政策措施，促进农村经济全面发展。完成农林牧渔业总产值15.01亿元，增长7%。粮食呈恢复性增长，播种面积76.92万亩，比上年增加5.68万亩，总产量增加3.27万吨。农业结构加快调整，发展经作22.7万亩，其中收购烤烟5.75万担，创税572万元，增长106.5%，实现产量、质量、效益和创税四项新高，荣获全市烤烟生产先进单位；种植食用菌8500万袋，增长41.6%。农业产业化稳步推进，新建、扩建绿之星食品、旭禾米业等29家加工龙头，仙楼竹制品、盛龙食品成长为省级龙头，“望丰精米”在全市首获省名牌产品，天凌纯奶、康安茶叶等通过“无公害农产品”标志论证；规模加工企业发展到110家，完成销售产值5.4亿元，增长12.5%，带动农户发展基地95.49万亩，完成农户订单合同和原料收购1.11亿元。转移农村劳动力5.92万人，实现创收2.96亿元，人均创收885元，成为全省典型。完成水利水电工程投资8170万元，农村生产生活条件进一步改善。
 三是旅游等第三产业加快发展。坚持以工业化推动第三产业发展，旅游和现代服务业实现新的突破。总投资300万美元溪下绿洲山水休闲旅游项目初步完成规划设计，动工兴建相关基础设施；总投资1亿元浮盖山风景区旅游道路等设施建设已启动；匡山、九石渡等景区配套设施建设加快。城市建设和经营成效明显。累计投资城市开发1.5亿元，基本完成怡源A、C、D区及浦城大酒店、新华书店城等城建重大项目，河滨路二期旧改完成投资6400多万元，城市建成区面积达8.45平方公里；全年收储土地75.84亩，出让土地4.6公顷，增加收益918万元。现代商贸流通不断活跃。新建、扩建专业和综合市场1.23万平方米，成立行业连锁经营企业7家，物流配送中心5家，新增大型超市7家；流通队伍发展到6万余人。第三产业对经济增长的贡献率达37.5%。
 四是固定资产投资规模明显扩大。认真实施项目带动战略，在全县上下深入开展以项目为核心的创业竞赛，推动投资平稳增长。全县落实项目169项，计划总投资15.76亿元，当年投资规模9.13亿元，完成投资7.27亿元。民间投资增长27%，占全社会固定资产投资的51.8%。总投资1000万元以上项目达23项，500-1000万元17项，300-500万元15项。“六大工程”、“十大项目”实施顺利。“百公里”农村路网工程建成177.5公里，其中县道52公里、乡道125.5公里，实现4个乡镇、56个村通乡公路，超额完成市下达任务；浦南高速公路前期工作扎实开展；城区电视网络升级改造、牲畜定点屠宰场、教育信息网、农村中小学危房改造、传染病房、体育健身中心均已基本完成。市重点项目华瑞化工硫酸、氢氟酸、1500千瓦余热发电、3.5万千伏变电站建设如期完成；绿康生化杆菌肽锌一、二期工程正式投产；南浦生态工业园区二期、正大生化浦潭工业新区及观前、蒋溪口电站建设有序展开。
 五是改革开放有效推进。全面落实免征农业税、缓征部分农业特产税、停征屠宰税等政策，其中免征农业税使农民受益近1200万元，税改配套改革步伐加快，成果不断扩大；集体林权制度改革扎实推进，全县发（换）林权证6574本，面积89.2万亩；农村工作机制进一步创新，县直89个部门与77个村开展“互动联动”，首批288名村级农技员竞聘上岗；农村信用社改革有序展开。国有集体企业改革不断深化，县粮食、供销、林业等所属企业改革基本完成，新改制县医药公司等6家企业，置换职工身份600余人。县乡财政管理体制继续完善，大幅调减乡镇收入基数，并实行更多的倾斜扶持，进一步激发了基层发展活力。着手实施事业单位人事、用工、分配制度改革。对外经贸迈出新步伐。完成新批外资项目9个，签约山海协作项目47个，利用县外资金4.35亿元，增长39%，其中来自江浙沪项目投资占63%，区域经济协作更趋紧密。
 六是各项社会事业协调发展。全年实施省、市科技项目7项，推广应用科技成果和开发新产品42项，科技对经济增长的贡献率达50%。在“6.18”省项目成果交易会上促成44个项目成果与企业对接，净增22项，荣获全市“6.18”项目成果推介工作先进集体；教育“两基”成果巩固提高，创办“闽北职业技术学院浦城分校”，全面实施小学和初中基础教育课程改革，学校德育建设不断加强，高中阶段教育加快发展，高考上线率位居全市第三。狠抓计生“八个突破”，新的计生机制初步形成。公共卫生防疫体系建设得到加强，新型农村合作医疗制度试点启动实施，新建仙阳、枫溪2所农村卫生院，农村医疗水平逐步改善。文体广电事业取得新进展，完成21项公益文化推介项目，成功举办全县第四届运动会，完成19个乡镇184个村三级电视网络联网，《浦城年鉴》正式出版发行，地方志工作获全省表彰。生态建设和环境保护稳步推进，“一控双达标”成果扩大，环境质量进一步改善。严格基本农田保护，加大土地整理和开发，连续多年实现耕地占补平衡。
 七是城乡居民生活进一步改善。为民谋利取得新成效。新增城镇就业岗位2213个，净增就业人员1218人，下岗失业人员再就业1188人，其中“4050”人员就业767人，实行社保援助1079人。社会保障体系不断完善，养老、失业、医疗保险及城乡居民最低生活保障覆盖面继续扩大，实现农村低保7563户12802人的应保尽保。建成1个乡级、43个村级供水工程，解决2.45万农民饮水困难，实施“造福工程”搬迁300人。支持和组织全社会助学、助残、助困和挂钩帮扶等活动。“餐桌污染”治理和食品“放心工程”建设力度加大，农村药品监管网络建设初具规模，市场整规工作成效明显。全年确定10件为民办实事项目顺利实施。汽车、住房、通讯等消费热点有所增长，居民消费价格总指数增长3.9%。

 二、坚持科学发展观扎实做好2005年工作
 2005年是全面实现“十五”计划目标，也是衔接“十一五”发展的关键一年。政府工作的总体要求是：以邓小平理论和“三个代表”重要思想为指导，认真贯彻党的十六届三中、四中全会和省、市经济工作会议精神，落实省委七届七次、八次全会和市委二届八次、九次全会的部署，全面落实科学发展观，坚持发展第一要务，更好利用重要战略机遇期，落实宏观调控政策，主动融入全省海峡西岸经济区和全市绿色腹地建设，深化改革开放，调整经济结构，转变增长方式，全面推进南接北连的生态经济走廊建设。继续实施项目带动、突出工业、狠抓“三农”、做大第三产业、发展县域经济、加快产业集聚，着力提升民营经济，转变政府职能，改善人民生活，维护社会稳定，构建和谐社会，实现经济社会全面协调可持续发展。
 2005年经济社会发展的预期目标是：生产总值增长10.5%，力争更快更好些；全社会固定资产投资增长18%；地方级财政收入同比增长10%；社会消费品零售总额增长12%；实际利用外资增长10%，外贸出口增长10%；城镇居民人均可支配收入增长7%；农民人均纯收入增长7%；城镇登记失业率控制在6% 以内；人口自然增长率控制在7‰以内。
 实现上述目标，必须自觉以科学发展观统领全局，强化危机和责任意识，继续突出项目、突出工业、突出开放、突出改革、突出为民，实现统筹兼顾，协调发展，把政府各项工作落到实处。

 （二）加快推进工业化进程，进一步扩大经济总量
 培育产业集群。依托优势产业和企业，集中各类生产要素，争取用3-5年时间，扶持壮大生物化工、林产加工、绿色食品、轻工、旅游等5个产值超10亿元的产业集群。生物化工产业以正大、绿康、绿安为重点，不断开发新产品，拓展新市场；林产加工产业以丽浦、仙楼为重点，积极采用新技术、新设备，提升产品技术含量和附加值；绿色食品产业以绿之星、藤本山、盛龙为重点，大力发展配套基地，扩张企业规模；轻工产业以合顺轻纺、闽城、京龙光学眼镜、正发服饰为重点，延伸中下游项目，形成相互配套的产业链条；旅游产业以浮盖山、溪下绿洲、匡山、九石渡旅游开发为重点，尽快生成效益。
 壮大规模工业。围绕产业集聚，拉长优势企业产业链，加大对现有骨干企业扶持服务力度，加快扩张现有45家规模以上工业企业，突出抓好正大浦潭工业新区、华瑞化工水泥粉磨站等重大扩建项目建设，尽快做大正大、华瑞、绿康、丽浦、绿安等一批骨干企业，增强支撑带动作用，催生产业链条，做到发展一个带动一批。大力引进一批高新项目和产业空白项目，培育升级5家潜力大、市场前景好、关联度强的规模以下企业，争取全年规模以上工业企业发展到50家以上，实现产值增长20%以上。
 拓展平台建设。继续完善园区一期配套工程，科学规划新一轮工业平台，加快园区二期建设，不断拓展发展空间。创新园区管理和运行机制，强化服务和管理功能，进一步改善投资环境，促进10家投产企业尽快达产达效，世通电子、宏远钢构、旭禾米业、京龙光学眼镜等13家在建、新建企业尽快建成投产，力争实现园区产值3亿元以上。提高入园企业标准和档次，大力推进招商引资，力争全年新引进10家以上特色产业大项目，逐步形成若干个关联企业集中、协作配套密切的“板块区”。
 发展民营经济。鼓励支持民营经济发展，实施“回归工程”，积极引导民营经济参与国企改造嫁接，投资基础设施、公用事业、服务业及其他行业和领域，激活民间资金。落实扶持民营经济各项政策措施，改进服务和监管方式，建立民营企业投资公司和行业担保公司，跟踪培育一批发展前景好的重点民营企业，引导帮助申领进出口经营权，加快制度、技术和管理创新，开展专业化协作和分工合作，实现集约和规模经营。
 （三）加强农业综合生产能力建设，保持农民收入持续增长
 培育壮大主导产业。稳定粮食产业，落实订单售粮直补政策，全面实施粮食“种子工程”，加强粮食流通管理，扶持望丰、旭禾等粮食龙头企业发展，提高粮食综合生产能力和种粮效益。围绕做大优质稻、畜禽业、木竹业、菌菜、烤烟五大优势产业，加快实行农业标准化生产，大力发展绿、特、名、优产品，发展一批专业乡、专业村，培育3-4个优质品牌。全年发展优质稻30万亩以上；经济作物22万亩，力争实现8万担烤烟种植，带动农业结构调整，提高产业整体素质。巩固乳牛业，扩大白鹅、肉兔、稻田鱼、土鸡等特色畜禽水产养殖规模，突出良种繁育、疫病防疫和污染治理三个关键，全面推开村级动物防疫员选聘工作，提高禽畜业规模效益。挖掘山地资源潜力，强化科技兴林，推广良种选育和高效竹林、经济林等科学种植技术，积极发展珍稀树种、工业原料林、毛竹丰产高效、大径材“四大基地”3.5万亩，推动林业资源的精深加工。

 加大政策支持力度。加快实施农村水利“523”工程，完成45个村级供水工程，解决2.5万人饮水困难，建成山地蓄水池118个，除险加固6座小（二）型水库，推进临江中型节水灌溉和富满潭水库工程建设，加强小型水利建设，建立商品粮基地和标准农田6000亩。继续实施年“百公里”农村公路改造，完成1个乡镇、29个建制村水泥路面硬化，其中县道（临江—枫溪）33公里、乡道71公里。开设竹山机耕道110公里。健全防汛抗旱等防灾、减灾体系，建立重大动植物疫病防控体系和森林防火长效机制，提高抵御自然灾害的能力。按照“民办、民管、民受益”的原则，引导农民发展完善各类中介服务组织，继续搞好村级农民技术员的管理和聘任。认真实施农村劳动力转移“阳光工程”，完成就业培训2000人，加强劳务派遣和跟踪服务，力争新增转移6000人。
 （四）加快发展第三产业，提升服务业整体素质
 加大旅游产业开发力度。按照旅游开发总体规划，加强旅游资源整合，突出景区保护、景点建设、要素配置、对外宣传四大重点，认真抓好溪下绿洲山水休闲旅游项目开发，全面完成路、电、讯等基础设施和绿化工程建设，力争2006年春节前第一期服务项目对外营业；加快浮盖山景区基础设施建设，争取今年国庆前有明显进展；加快匡山国家级森林公园的申报，推进匡山、九石渡景区的对外招商开发，力争年内有实质性突破；充分挖掘观前省级历史文化名村、仙楼山、梦笔山、西山故里等人文旅游资源，丰富旅游内涵。积极发展旅游企业，加强区域合作，扩大旅游影响。
 建设现代流通业。抓好工业园区、浮盖山景区商业网点规划建设，加快发展城区物流配送、电子商务、连锁经营等现代流通业态，鼓励引导现代物流企业创立农村网点，扶持建立笋、菇、茶、柰、白鹅等一批特色专业市场，加强流通队伍建设，进一步发展和规范行业协会，提高农村流通组织化水平。积极发展社区便民服务店和餐饮、会计、审计、资产评估、经纪与代理等各类中介服务机构，推进房地产、金融、信息、技术等现代服务业。深入整顿和规范市场经济秩序，重点加强对食品、药品及房产交易市场的监管，保护知识产权，严厉打击制假售假、商业欺诈等违法行为。
 加快中心城镇发展。围绕做大做强中心县城，加快实施河滨路二期、怡源D区、莲花垅小区等旧改和新区建设，力争贯通江滨路全线。组建城市资产运营中心，提高城市经营水平，逐步完善城市功能。健全落实城市长效管理机制，着力整治违章建筑和环境脏乱差等群众关注的热点难点问题。全面完成205国道境内路面改造，做好浦南、宁上等高速路网沿线的石陂、临江、莲塘、仙阳等中心集镇的用地规划、调整和报批等工作，加快发展互通口园区经济，逐步形成南浦生态工业、临江食品加工、石陂竹木加工、浦潭精细化工和仙阳边贸五大南北连线、东西呼应的“经贸小区”，吸引人流、物流、信息流向中心城镇集聚，增强聚集和辐射功能，推进城镇一体化进程。
 （五）着力推进各项改革，激发体制机制活力
 深化国有集体企业改革。加大尚未改制的国有集体企业重组盘活力度，全面核实县林化厂、一化厂、印刷厂、闽北制药厂等国有工业企业资产、负债等情况，适时整体捆绑协商解决银行债务，积极引入民营、外资参与改组改制，力争年内基本完成改革任务；稳步推进二轻、建筑等所属企业改革，基本完成电力企业股份制改造，并妥善解决企业改制后的遗留问题。加强国有资产监管，防止流失，确保保值增值。
 深化农村各项改革。扎实推进集体林权制度改革，争取年底基本完成改革目标。深化林改后的配套改革，组建县林业服务中心、行业协会和专业合作经济组织，加快建立林业经营管理新体制。继续创新农村工作机制，深入开展“互动联动”和“五强五星”活动，提高“五支下派队伍”服务水平。抓好农村税改后的各项配套改革，全面取消农业税及其附加和除烟叶特产税外所有农业特产税，加强农民负担监督管理，稳定农村土地家庭承包制度，严格土地征用和保护，有序引导土地规模流转，保障农民合法权益。
 深化财税金融体制改革。积极培植财政收入新增长点，依法加强管征，逐步建立收入稳定增长的机制；深化“收支两条线”管理，严控一般性支出，6月底将全县行政事业单位纳入县会计中心实行集中核算，进一步深化和完善政府采购制度，集中财力保障人民群众切身利益和重点支出；调整完善县乡财政管理体制，全面实行乡镇财政供养人员工资县级国库统一发放，严禁截留、挪用村级干部工资，逐步解决乡镇干部社保、医保问题；继续实行乡镇新增增值税县得部分全部或大头留给乡镇，烤烟特产税全额返还乡镇的奖励政策，将更多新增财力留给基层，充分调动乡镇抓发展的积极性。全面深化农村信用社改革，积极开展农村信用户、信用村和信用乡镇创建活动；支持国有商业银行股份制改革，健全政银企会商制度，严厉打击逃废银行债务，扩大金融信贷对中小企业的投放量，发挥服务地方经济作用。
 深化行政管理体制改革。按照全省统一部署，完成政府机构改革，加快建立与服务型政府相适应的审批制度，进一步转变政府职能。分类实施外贸、物资、水利水管等事业单位改革，加快自收自支事业单位转企撤并。稳步推进社区建设，制定出台撤镇设街和社区整合方案，年底前全面完成社区整合工作。加快乡镇综合改革，合理设置乡镇机构，分流人员，减少财政供养人员，做好削赤减债工作。
 （六）积极扩大对外开放，大力发展外向型经济
 注重招商实效。着力在实效性上下功夫，广泛搜集筛选信息，精心论证招商项目，大胆创新招商方式，推行专业招商、以商招商、“回归”招商、委托招商和网络招商，以企业为主体，发动全社会力量，重点引进一批产业集群的关键项目和缺失项目及工农业开发、农产品加工的外资项目，推动产业链群招商，不断扩大对内对外开放。精心组织参加“5.16”、“6.18”和“9.8”等各类经贸洽谈会，加强对签约项目的全程跟踪服务，着力提高项目开工率、到资率和履约率，更多更好地利用外资。
 扩大出口贸易。抓住省上出口退税政策维持不变的有利时机，大力扩展对外贸易。在巩固提升饲料金霉素、盐酸金霉素等重点传统出口产品的基础上，扩大竹木产品、笋制品、食用菌、水果罐头、制衣、眼镜等产品出口规模。深度开发传统出口市场，不断开拓新兴国际市场，鼓励更多企业争取自营外贸出口权，多渠道扩大外贸出口，确保出口稳定增长。
 改善优化环境。树立“人人都是投资环境，人人都是浦城形象”观念，真正形成改善环境的共识共为，以开放的理念倾力打造更加宽松的发展环境。围绕放低“行政门槛”、“经济门槛”，组建行政服务中心，形成项目引进时
“一站式”服务、建设过程中全方位服务、开工投产后经常性服务的运作机制，提高办事效率。研究完善吸引投资的政策措施，努力降低投资成本，在政策允许范围内保护和让投资商得到最大满足、获得最大利益。严格责任追究，严肃查处破坏投资环境的人和事。
 （七）切实维护人民利益，办好惠及百姓的实事
 努力扩大就业。突出在统筹发展、突破工业中增加就业，在落实政策、具体运作中促进就业，在改变观念、创新举措中实现就业，多渠道增加就业岗位；健全再就业援助制度，建立和完善就业、培训、保障、维权相结合的就业服务体系，重点帮助“4050”人员、下岗失业人员再就业，力争全年新增就业岗位3900个，其中 “4050”人员再就业800人。
 完善社会保障。多渠道筹集社会保障基金，继续扩大养老、失业、工伤及医疗保险的覆盖面，将水北街、石陂、永兴、忠信、九牧、管厝6个乡镇的1589名教职工新纳入医保。进一步完善规范城镇和农村低保工作，实行农村低保家庭子女接受义务教育“两免一补”制度，全面实施农村部分计划生育家庭奖励扶持政策，推进社会保障体系向农村拓展。
 加强扶贫解困。坚持以人为本，着力解决涉及群众切身利益的突出问题，千方百计提高城乡居民生活。进一步改善革命老区、少数民族村的生产生活条件，继续实施“五通”建设和“造福工程”，加大开发性扶贫力度，抓好农村改旧建新工作，努力改善农村人居环境。高度关注城乡困难群体，认真安排好下岗职工、困难企业职工、城市贫困人口、受灾群众、贫困残疾人和老年人等生活。加强农民工工资支付的监管，严禁新欠，清理旧欠，维护农民工合法权益。
 （八）坚持经济社会统筹协调，实现可持续和谐发展
 加强精神文明建设。认真贯彻《公民道德建设实施纲要》，切实抓好未成年人的思想道德建设，加强社会公德、职业道德、家庭美德和诚实守信教育，深入开展文明县城、文明行业、文明村镇、文明社区创建及区域联片共建活动，进一步提高省级文明县城创建水平。弘扬爱国精神、创业精神和社会正气，形成扶正祛邪、扬善惩恶的社会风尚。普及科学知识，反对封建迷信，不断提高城乡文明程度。
 加快科技、人才和教育等事业发展。加快科技进步与创新，积极申报4项省、市科技项目，抓好40项科技成果推广应用和新产品开发。大力实施人才战略，努力构建人才引进、培养和使用机制，全员分配医学和师范类院校本科毕业生，继续为100名中高级职称教师办理职称进岗，优化完善人才环境。重视幼儿教育，巩固发展农村九年义务教育，扶持浦一中办成优质高中，大力发展职业技术教育，积极发展民办教育。加强学校思想道德建设、教师队伍建设和教育行风建设。大力挖掘民间民俗文化资源，重视艺术精品创作和艺术人才培养，切实保护好省、县级文物单位和文化遗产。加强文体基础设施建设，广泛开展全民健身活动，提高群众性体育水平。深化医疗卫生体制改革，加快建立疾病信息网络、预防控制、医疗救治和执法监督等公共卫生体系，推进社区卫生服务所、乡镇卫生院建设，积极发展民营医院，抓好新型农村合作医疗制度试点工作，继续推进农村药品监管网络建设，启动并加快农村药品供应网络建设，解决群众因病返贫致贫问题。完成20个乡村广电联网改造。
 推进人口资源和环境保护工作。依法治理人口环境，狠抓人口与计生工作整改，创新计生工作理念、机制和方法，加大政策推动和宣传力度，突出解决出生人口性别比、流动人口管理、社会抚养费征收等问题，确保政策符合率达到省、市要求，扭转计生被动局面。加强土地管理，实行最严格耕地、林地保护制度，做好土地利用总规修编，提高土地集约经营。全面推进国家级生态示范区建设，落实环保目标责任制，加大工业企业污染和南浦溪流域的综合整治。加强地质灾害防治，积极做好枫溪下胡推省重点地灾点群众搬迁安置工作。推广农村沼气工程建设，鼓励全社会节约资源，减少资源消耗，改善环境质量。
 深入创建“平安浦城”。落实社会治安综合治理责任制，加快防控体系建设，保持对严重刑事犯罪活动的严打高压态势，重点打击有组织犯罪、暴力犯罪和多发性侵财犯罪。认真受理群众信访，依法及时解决涉及山林权属、征地拆迁等群众反映强烈的问题，全面推进“餐桌污染”治理和食品“放心工程”建设，努力维护社会公平和正义。落实安全生产和重大安全事故一把手责任制，开展安全生产经常性检查，及时有效排除隐患，坚决遏制重特大事故。制定和落实处置经济社会紧急情况工作预案，提高保障公共安全和处置突发事件的能力，全力维护社会安定稳定。
 三、切实加强政府行政能力建设
 执政为民，字字千钧，责任如山。政府各部门要进一步增强责任感和使命感，加快管理制度、方式和方法创新，致力提升执政为民能力，切实履行好人民赋予的重任。
 转变职能科学行政。运用经济和法律手段改进政府管理经济方式，在继续搞好经济调节、市场监管的同时，创新公共服务机制，强化政府社会管理和公共服务的职能，切实把该管的事管好管到位，把不该管的事放开放活。完善政府重大经济社会问题集体决策制度和决策责任制，逐步规范咨询、论证、审议、公布、备案和听证等程序，强化各项监督，提高决策科学化、民主化、法治化。健全完善政府学习制度，提高公务员行政能力，加快建设学习型政府。
 公正透明依法行政。带头维护宪法和法律权威，继续抓好“四五”普法，认真落实《行政许可法》和《全面推进依法行政实施纲要》，推进配套制度建设，完善行政复议制度，严格依法行政，加快建设法制型政府。规范行政执法行为，严格按照法定权限和程序行使权力，履行职责，做到依法办事、依法决策、依法处理问题，严格执法、公正执法、文明执法，促进政府行为的法治化、规范化。纵深推进机关效能建设，深化政务公开，切实增强行政透明度，提高政府为民办事效能。
 接受监督民主行政。自觉接受人大及其常委会的监督，主动向人大及其常委会报告工作，认真执行人大决议、决定，积极配合人大开展执法检查和人大代表视察。积极支持政协履行政治协商、民主监督和参政议政的职能，完善政府与政协工作联系制度。及时认真办理人大代表、政协委员建议、提案。认真听取各民主党派、工商联、无党派人士及专家学者的意见，充分发挥工会、共青团、妇联等群团组织的桥梁纽带作用，广泛接受社会各界及群众监督。扎实推进基层民主政治建设，扩大基层民主。
