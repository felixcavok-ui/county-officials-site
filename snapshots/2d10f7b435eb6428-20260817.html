function setCookie(name, value, time) {
 var strsec = getsec(time);
 var exp = new Date();
 exp.setTime(exp.getTime() + strsec * 1);
 document.cookie = name + "=" + escape(value) + ";path=/;expires=" + exp.toGMTString();
}

function getsec(str) {
 var str1 = str.substring(1, str.length) * 1;
 var str2 = str.substring(0, 1);
 if (str2 == "s") {
 return str1 * 1000;
 } else if (str2 == "h") {
 return str1 * 60 * 60 * 1000;
 } else if (str2 == "d") {
 return str1 * 24 * 60 * 60 * 1000;
 }
}

function getCookie(name) {
 var arr, reg = new RegExp("(^| )" + name + "=([^;]*)(;|$)");
 if (arr = document.cookie.match(reg)) return unescape(arr[2]);
 else return null;
}

function delCookie(name) {
 var exp = new Date();
 exp.setTime(exp.getTime() - 1);
 var cval = getCookie(name);
 if (cval != null)
 document.cookie = name + "=" + cval + ";path=/;expires=" + exp.toGMTString();
}

 　　2月24日上午，光泽县政协第九届委员会第五次会议开幕。下午4点30分，参加完几个组讨论的县长赵明正就在政府会议室，就《扶贫莫忘了被划企业的村庄》提案，研究解决办法。 
 “上午开会时，我看了《县政协重点提案摘要》，发现我县还有20多个点、2000多大集体时被划企业的村庄和农民，至今还没享受到农村强农政策，成为精准扶贫的遗忘群体，心里很急。”赵明正说，“所以，把八个乡（镇）的乡（镇）长和民政、财政、农业、林业等相关部门的领导叫来，研究一个解决办法。” 
 解决被划企业村庄的归村问题，是各乡（镇）和相关部门一直纠结、闹心的难题。各乡（镇）长先后汇报了存在情况和问题，也谈了各自的解决建议，相关部门也从归村报批程序、财政归盘、资产处置等方面谈了难度和建议。最大问题归结到两点：归村建制不难，县里就能解决。但归村后的财政转移支付，一事一议和各项以村为单位的基础设施建设项目等，都得进入省发改和财政部门的盘子，归村才算解决问题。还有被划企业的村庄规模有大有小，大的可建村，小的归村、合并也存在资金、资产、资源处置等系列问题，怎么办？ 
 “正因为难度大，所以问题才拖到现在。到2020年全面建成小康社会，不能因为要入省里的盘子就等，更不能因为困难大就无所作为。”赵明正县长说，“时不我待。我们现在要做的，一是先把设村‘挂号’的事抓紧上报，同时把我们自己能做得事抓紧做好。” 
 经过集思广益，会议形成以下解决办法：对没有土地、林地等产权纠纷，户数在100户以上，可设村的先抓紧报批；原来就由乡（镇）管辖的被划企业，能归原划出村的“哪来哪去”；原村归不了，合并合不了的，由乡（镇）直接管，政治待遇、经济待遇参照村级，先由县、乡（镇）给付，上级来的项目一视同仁、共享。最后赵明正强调：“做好被划企业共享小康成果的工作，一要强化属地责任，各家孩子各家抱。二要鼓励创新，敢于担当。不能等上头文件再动，要开动脑筋，把被划企业村庄和农民的脱贫致富，建成小康的事做好。决不能漏掉一个村、落下一户人。” 
 会上，赵明正县长还当场交待江晖副县长牵头抓落实。(邱盛林)
