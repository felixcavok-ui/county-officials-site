function setCookie(name, value, time) {
 var strsec = getsec(time);
 var exp = new Date();
 exp.setTime(exp.getTime() + strsec * 1);
 document.cookie = name + "=" + escape(value) + ";path=/;expires=" + exp.toGMTString();
}
function getsec(str) {
 var str1 = str.substring(1, str.length) * 1;
 var str2 = str.substring(0, 1);
 if (str2 == "s") {
 return str1 * 1000;
 } else if (str2 == "h") {
 return str1 * 60 * 60 * 1000;
 } else if (str2 == "d") {
 return str1 * 24 * 60 * 60 * 1000;
 }
}
function getCookie(name) {
 var arr, reg = new RegExp("(^| )" + name + "=([^;]*)(;|$)");
 if (arr = document.cookie.match(reg)) return unescape(arr[2]);
 else return null;
}

 在中国共产党成立105周年之际，浦城县“两优一先”表彰大会召开，激励动员全县各级党组织和广大党员踔厉奋发、勇毅前行，奋力推动浦城绿色高质量发展。县委书记陈锡明出席会议并讲话。县委副书记、县政府党组书记叶财旺主持会议。县人大常委会主任杨文龙、县政协主席郑辉出席。县委常委、组织部部长刘超宣读表彰决定。 陈锡明在讲话中代表县委向受表彰的先进集体和个人表示热烈祝贺，向长期以来为浦城县党的建设和各项工作作出重要贡献的广大党员、党务工作者致以诚挚问候。他强调，全县各级党组织和广大党员干部要认真学习贯彻习近平新时代中国特色社会主义思想，深入贯彻落实习近平总书记对福建、对南平工作的重要讲话重要指示批示精神，传承“四下基层”“马上就办、真抓实干”等优良作风，弘扬廖俊波同志先进事迹，深入开展“三争”行动，对标先进、见贤思齐，奋力谱写新征程新浦城建设新篇章。 陈锡明强调，要铸牢忠诚之魂，在坚定理想信念中把准政治方向，坚持不懈用习近平新时代中国特色社会主义思想凝心铸魂，深入学习贯彻党的二十大和二十届历次全会精神，坚持读原著、学原文、悟原理，做到学思用贯通、知信行统一，切实把学习成果转化为坚定拥护“两个确立”、坚决做到“两个维护”的高度自觉。要勇于担当实干，在绿色高质量发展中当好先锋表率，全力推动传统产业转型升级，持续推进浦城经济开发区建设，发挥福大国家大学科技园等平台作用，做优绿色食品加工、生物制造等产业链条，抓好幸福里、“4＋N”等民生领域改革，坚持和发展新时代“枫桥经验”，推动矛盾纠纷发现在早、化解在小。要夯实执政之基，在全面从严治党中筑牢战斗堡垒，深入开展树立和践行正确政绩观学习教育，把学习贯彻习近平党建思想作为当前和今后一个时期的一项重要政治任务，深化实施新时代“堡垒工程”，深入开展基层党建“创优争先年”活动，常态化、精准化、差异化开展农村党建“体检”，高标准高质量做好村（社区）“两委”换届，深化“片区联建共富”机制，推动县乡两级振兴公司合作共赢，统筹推进各领域党建工作，以高质量党建引领保障浦城绿色高质量发展。 会上，县领导为受表彰的“两优一先”代表颁奖，受表彰对象代表作了发言。 县委、县人大常委会、县政府、县政协领导，县法院院长、县检察院检察长出席。县直单位党组（党委、总支、支部）及省、市驻浦单位党组（总支、支部）主要负责同志，各乡镇（街道）党（工）委主要负责同志、组织委员；全省优秀共产党员、南平市第九批“廖俊波式好党员干部”和全县“两优一先”代表等参加。　　 （柯聪 柳浪漫）
